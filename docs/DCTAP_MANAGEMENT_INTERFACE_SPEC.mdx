## Executive Summary

This specification defines a comprehensive DCTAP (Dublin Core Tabular Application Profiles) Management Interface for the IFLA Standards platform. DCTAP serves as the foundational schema definition system that generates Zod contracts, TinaCMS form schemas, CSV profiles, and TypeScript types. This interface enables visual creation, editing, and management of application profiles that define the structure and validation rules for IFLA standards elements and vocabularies.

## Goals and Objectives

### Primary Goals
1. **Visual Profile Creation**: Intuitive interface for defining DCTAP application profiles
2. **Schema Generation**: Automatic generation of validation schemas and form definitions
3. **RDF Integration**: Import existing RDF schemas and export to multiple formats
4. **Multi-Site Support**: Manage profiles for all 6 IFLA standards sites
5. **Version Control**: Track profile changes with Git-based versioning
6. **Validation Framework**: Ensure profile consistency and compliance

### Success Criteria
- SMEs can create and modify application profiles without technical knowledge
- 100% automated generation of downstream schemas (Zod, TinaCMS, CSV)
- Import 95%+ of existing RDF element definitions accurately
- Zero-downtime profile updates with backward compatibility
- Complete audit trail for all profile changes

## DCTAP Overview

### What is DCTAP?
Dublin Core Tabular Application Profiles (DCTAP) is a specification for describing metadata application profiles in a simple tabular format. It defines:

- **Shapes**: Groups of properties that describe a resource type
- **Property Constraints**: Rules and restrictions for individual properties
- **Value Types**: Expected data types and formats
- **Cardinality**: Whether properties are mandatory, optional, or repeatable
- **Value Constraints**: Controlled vocabularies, patterns, or validation rules

### DCTAP in IFLA Context
For IFLA Standards, DCTAP profiles define:
- **Element Definitions**: Properties, domains, ranges, and relationships
- **Vocabulary Structures**: Term definitions, hierarchies, and validation rules
- **Content Validation**: Rules for MDX front matter and content structure
- **Form Generation**: Schema for TinaCMS editing interfaces

#### 3. RDF Import/Export Engine
Handles bidirectional conversion between RDF and DCTAP formats.

```typescript
class RDFImportEngine {
  async importFromRDF(rdfContent: string, format: 'turtle' | 'rdf-xml' | 'json-ld'): Promise<DCTAPProfile[]> {
    // Parse RDF content
    const graph = await this.parseRDF(rdfContent, format);
    
    // Extract property definitions
    const properties = await this.extractProperties(graph);
    
    // Create DCTAP profile
    const profile: DCTAPProfile = {
      id: generateUUID(),
      shapeID: this.extractShapeID(graph),
      shapeLabel: this.extractShapeLabel(graph),
      version: '1.0.0',
      namespace: this.extractNamespace(graph),
      prefix: this.extractPrefix(graph),
      description: this.extractDescription(graph),
      created: new Date().toISOString(),
      modified: new Date().toISOString(),
      createdBy: 'rdf-import',
      modifiedBy: 'rdf-import',
      properties: properties,
      siteKey: this.detectSiteKey(graph),
      profileType: this.detectProfileType(graph),
      generationConfig: this.getDefaultGenerationConfig(),
      status: ProfileStatus.DRAFT,
      published: false,
      deprecated: false,
    };
    
    return [profile];
  }
  
  private async extractProperties(graph: RDFGraph): Promise<DCTAPProperty[]> {
    const properties: DCTAPProperty[] = [];
    
    // Find all property definitions
    const propertyNodes = graph.getNodes({ type: 'rdf:Property' });
    
    for (const node of propertyNodes) {
      const property: DCTAPProperty = {
        propertyID: this.extractPropertyID(node),
        propertyLabel: this.extractLabel(node),
        propertyURI: node.uri,
        propertyDefinition: this.extractDefinition(node),
        
        // Extract constraints
        mandatory: this.extractMandatory(node),
        repeatable: this.extractRepeatable(node),
        valueDataType: this.extractDataType(node),
        valueConstraint: this.extractConstraint(node),
        
        // Extract relationships
        subPropertyOf: this.extractSubProperties(node),
        equivalentProperty: this.extractEquivalentProperties(node),
        inverseOf: this.extractInverse(node),
        
        // UI hints (generated based on data type)
        displayOrder: properties.length,
        uiHints: this.generateUIHints(node),
        validation: this.generateValidationRules(node),
        
        deprecated: this.isDeprecated(node),
      };
      
      properties.push(property);
    }
    
    return properties.sort((a, b) => a.displayOrder - b.displayOrder);
  }
  
  private generateUIHints(node: RDFNode): UIHints {
    // Infer UI field type from RDF properties
    const range = this.extractRange(node);
    const datatype = this.extractDatatype(node);
    
    if (range === 'xsd:boolean') {
      return { fieldType: FieldType.CHECKBOX };
    }
    
    if (range === 'xsd:date') {
      return { fieldType: FieldType.DATE };
    }
    
    if (datatype === 'xsd:anyURI') {
      return { fieldType: FieldType.URI };
    }
    
    // Check for controlled vocabulary
    const vocabulary = this.extractVocabulary(node);
    if (vocabulary) {
      return {
        fieldType: FieldType.SELECT,
        options: this.extractVocabularyOptions(vocabulary),
      };
    }
    
    // Default to text input
    return { fieldType: FieldType.TEXT };
  }
}

class RDFExportEngine {
  async exportToRDF(profile: DCTAPProfile, format: 'turtle' | 'rdf-xml' | 'json-ld'): Promise<string> {
    // Create RDF graph
    const graph = new RDFGraph();
    
    // Add namespace declarations
    graph.addNamespace(profile.prefix, profile.namespace);
    
    // Add shape definition
    const shapeURI = `${profile.namespace}${profile.shapeID}`;
    graph.add({
      subject: shapeURI,
      predicate: 'rdf:type',
      object: 'sh:NodeShape',
    });
    
    graph.add({
      subject: shapeURI,
      predicate: 'rdfs:label',
      object: profile.shapeLabel,
    });
    
    // Add property definitions
    for (const property of profile.properties) {
      await this.addPropertyToGraph(graph, property, profile);
    }
    
    // Serialize to requested format
    return await graph.serialize(format);
  }
  
  private async addPropertyToGraph(graph: RDFGraph, property: DCTAPProperty, profile: DCTAPProfile): Promise<void> {
    const propertyURI = property.propertyURI || `${profile.namespace}${property.propertyID}`;
    
    // Basic property definition
    graph.add({
      subject: propertyURI,
      predicate: 'rdf:type',
      object: 'rdf:Property',
    });
    
    graph.add({
      subject: propertyURI,
      predicate: 'rdfs:label',
      object: property.propertyLabel,
    });
    
    if (property.propertyDefinition) {
      graph.add({
        subject: propertyURI,
        predicate: 'rdfs:comment',
        object: property.propertyDefinition,
      });
    }
    
    // Add constraints as SHACL shapes
    if (property.mandatory === MandatoryValue.TRUE) {
      graph.add({
        subject: propertyURI,
        predicate: 'sh:minCount',
        object: '1',
        datatype: 'xsd:integer',
      });
    }
    
    if (property.repeatable === RepeatableValue.FALSE) {
      graph.add({
        subject: propertyURI,
        predicate: 'sh:maxCount',
        object: '1',
        datatype: 'xsd:integer',
      });
    }
    
    // Add relationships
    for (const subProp of property.subPropertyOf || []) {
      graph.add({
        subject: propertyURI,
        predicate: 'rdfs:subPropertyOf',
        object: subProp,
      });
    }
  }
}
```

#### 4. Code Generation Engine
Generates schemas and form definitions from DCTAP profiles.

```typescript
class ZodGenerator {
  async generateSchema(profile: DCTAPProfile): Promise<string> {
    const imports = ["import { z } from 'zod';"];
    const schemaFields: string[] = [];
    
    for (const property of profile.properties) {
      const fieldSchema = this.generateFieldSchema(property);
      schemaFields.push(`  ${property.propertyID}: ${fieldSchema}`);
    }
    
    const schemaName = `${profile.shapeID}Schema`;
    const schema = `
export const ${schemaName} = z.object({
${schemaFields.join(',\n')}
});

export type ${profile.shapeID} = z.infer<typeof ${schemaName}>;
`;
    
    return imports.join('\n') + '\n' + schema;
  }
  
  private generateFieldSchema(property: DCTAPProperty): string {
    let baseSchema: string;
    
    // Base type
    switch (property.valueDataType) {
      case ValueDataType.LITERAL:
        if (property.uiHints.fieldType === FieldType.TEXTAREA) {
          baseSchema = 'z.string()';
        } else if (property.uiHints.fieldType === FieldType.DATE) {
          baseSchema = 'z.string().datetime()';
        } else {
          baseSchema = 'z.string()';
        }
        break;
      case ValueDataType.URI:
        baseSchema = 'z.string().url()';
        break;
      default:
        baseSchema = 'z.string()';
    }
    
    // Add constraints
    if (property.validation?.minLength) {
      baseSchema += `.min(${property.validation.minLength})`;
    }
    
    if (property.validation?.maxLength) {
      baseSchema += `.max(${property.validation.maxLength})`;
    }
    
    if (property.validation?.pattern) {
      baseSchema += `.regex(/${property.validation.pattern}/)`;
    }
    
    if (property.valueConstraint) {
      // Handle controlled vocabulary
      if (property.uiHints.options?.length) {
        const values = property.uiHints.options.map(opt => `"${opt.value}"`).join(', ');
        baseSchema = `z.enum([${values}])`;
      }
    }
    
    // Handle arrays (repeatable)
    if (property.repeatable === RepeatableValue.TRUE) {
      baseSchema = `z.array(${baseSchema})`;
    }
    
    // Handle optional fields
    if (property.mandatory === MandatoryValue.FALSE) {
      baseSchema += '.optional()';
    }
    
    return baseSchema;
  }
}

class TinaCMSGenerator {
  async generateCollection(profile: DCTAPProfile): Promise<string> {
    const collectionName = this.toCamelCase(profile.shapeID);
    
    const fields = profile.properties.map(property => this.generateField(property));
    
    const collection = `
export const ${collectionName}Collection: Collection = {
  name: '${profile.shapeID.toLowerCase()}',
  label: '${profile.shapeLabel}',
  path: 'standards/${profile.siteKey}/docs/${this.getPathFromProfileType(profile.profileType)}',
  format: 'mdx',
  fields: [
${fields.map(field => '    ' + field).join(',\n')}
  ],
};`;
    
    return collection;
  }
  
  private generateField(property: DCTAPProperty): string {
    const fieldConfig: any = {
      type: this.mapFieldType(property.uiHints.fieldType),
      name: property.propertyID,
      label: property.propertyLabel,
    };
    
    if (property.mandatory === MandatoryValue.TRUE) {
      fieldConfig.required = true;
    }
    
    if (property.uiHints.helpText) {
      fieldConfig.description = property.uiHints.helpText;
    }
    
    if (property.repeatable === RepeatableValue.TRUE) {
      fieldConfig.list = true;
    }
    
    // Add options for select fields
    if (property.uiHints.options?.length) {
      fieldConfig.options = property.uiHints.options;
    }
    
    // Add validation
    if (property.validation || property.valueConstraint) {
      fieldConfig.ui = {
        validate: this.generateValidationFunction(property),
      };
    }
    
    return JSON.stringify(fieldConfig, null, 2);
  }
  
  private mapFieldType(fieldType: FieldType): string {
    switch (fieldType) {
      case FieldType.TEXT: return 'string';
      case FieldType.TEXTAREA: return 'string';
      case FieldType.SELECT: return 'string';
      case FieldType.CHECKBOX: return 'boolean';
      case FieldType.DATE: return 'datetime';
      case FieldType.URI: return 'string';
      case FieldType.RICH_TEXT: return 'rich-text';
      case FieldType.REFERENCE: return 'reference';
      default: return 'string';
    }
  }
}

class CSVGenerator {
  async generateProfile(profile: DCTAPProfile): Promise<string> {
    const csvProfile = {
      name: profile.shapeLabel,
      version: profile.version,
      description: profile.description,
      columns: {} as Record<string, any>,
    };
    
    for (const property of profile.properties) {
      csvProfile.columns[property.propertyLabel] = {
        required: property.mandatory === MandatoryValue.TRUE,
        datatype: this.mapDataType(property.valueDataType),
        constraint: property.valueConstraint,
        description: property.propertyDefinition,
        repeatable: property.repeatable === RepeatableValue.TRUE,
      };
    }
    
    return JSON.stringify(csvProfile, null, 2);
  }
  
  private mapDataType(dataType: ValueDataType): string {
    switch (dataType) {
      case ValueDataType.LITERAL: return 'string';
      case ValueDataType.URI: return 'uri';
      case ValueDataType.BNODE: return 'object';
      default: return 'string';
    }
  }
}

class TypeScriptGenerator {
  async generateTypes(profile: DCTAPProfile): Promise<string> {
    const interfaceName = profile.shapeID;
    const properties: string[] = [];
    
    for (const property of profile.properties) {
      const optional = property.mandatory === MandatoryValue.FALSE ? '?' : '';
      const type = this.generateTypeScriptType(property);
      const comment = property.propertyDefinition ? 
        `  /** ${property.propertyDefinition} */\n` : '';
      
      properties.push(
        comment + `  ${property.propertyID}${optional}: ${type};`
      );
    }
    
    const interface_ = `
/**
 * ${profile.shapeLabel}
 * ${profile.description}
 */
export interface ${interfaceName} {
${properties.join('\n')}
}`;
    
    return interface_;
  }
  
  private generateTypeScriptType(property: DCTAPProperty): string {
    let baseType: string;
    
    if (property.uiHints.fieldType === FieldType.CHECKBOX) {
      baseType = 'boolean';
    } else if (property.uiHints.fieldType === FieldType.DATE) {
      baseType = 'Date | string';
    } else if (property.uiHints.options?.length) {
      const values = property.uiHints.options.map(opt => `'${opt.value}'`).join(' | ');
      baseType = values;
    } else {
      baseType = 'string';
    }
    
    // Handle arrays
    if (property.repeatable === RepeatableValue.TRUE) {
      baseType += '[]';
    }
    
    return baseType;
  }
}
```

```typescript
class ProfileService {
  async createProfile(template: string, overrides: Partial<DCTAPProfile>): Promise<DCTAPProfile> {
    // Load template
    const templateData = PROFILE_TEMPLATES[template];
    if (!templateData) {
      throw new Error(`Template not found: ${template}`);
    }
    
    // Create profile from template
    const profile: DCTAPProfile = {
      id: generateUUID(),
      ...templateData,
      ...overrides,
      created: new Date().toISOString(),
      modified: new Date().toISOString(),
      status: ProfileStatus.DRAFT,
      published: false,
    };
    
    // Validate profile
    await this.validateProfile(profile);
    
    // Save to database
    await this.saveProfile(profile);
    
    // Log activity
    await jobAPI.logActivity({
      type: ActivityType.CONTENT,
      source: ActivitySource.ADMIN,
      userId: getCurrentUser().id,
      action: ActivityAction.CREATE,
      description: `Created DCTAP profile: ${profile.shapeLabel}`,
      resourceType: 'dctap_profile',
      resourceId: profile.id,
      metadata: { template, profileType: profile.profileType },
    });
    
    return profile;
  }
  
  async generateSchemas(profileId: string): Promise<GeneratedSchemas> {
    const profile = await this.getProfile(profileId);
    
    // Create generation job
    const job = await jobAPI.createJob({
      type: JobType.CODE_GENERATION,
      title: `Generate schemas for ${profile.shapeLabel}`,
      userId: getCurrentUser().id,
      resourceId: profileId,
      payload: { profileId, generators: ['zod', 'tinacms', 'csv', 'types'] },
    });
    
    try {
      const schemas: GeneratedSchemas = {};
      
      if (profile.generationConfig.generateZod) {
        schemas.zod = await new ZodGenerator().generateSchema(profile);
      }
      
      if (profile.generationConfig.generateTinaCMS) {
        schemas.tinacms = await new TinaCMSGenerator().generateCollection(profile);
      }
      
      if (profile.generationConfig.generateCSV) {
        schemas.csv = await new CSVGenerator().generateProfile(profile);
      }
      
      if (profile.generationConfig.generateTypes) {
        schemas.types = await new TypeScriptGenerator().generateTypes(profile);
      }
      
      await jobAPI.updateJobStatus(job.id, JobStatus.COMPLETED);
      
      return schemas;
    } catch (error) {
      await jobAPI.updateJobStatus(job.id, JobStatus.FAILED);
      throw error;
    }
  }
  
  async publishProfile(profileId: string): Promise<void> {
    const profile = await this.getProfile(profileId);
    
    // Validate before publishing
    await this.validateProfile(profile);
    
    // Generate all schemas
    const schemas = await this.generateSchemas(profileId);
    
    // Write generated files to appropriate locations
    await this.writeGeneratedFiles(profile, schemas);
    
    // Update profile status
    profile.status = ProfileStatus.PUBLISHED;
    profile.published = true;
    profile.modified = new Date().toISOString();
    
    await this.saveProfile(profile);
    
    // Create Git commit
    const job = await jobAPI.createJob({
      type: JobType.GIT_SYNC,
      title: `Publish DCTAP profile: ${profile.shapeLabel}`,
      userId: getCurrentUser().id,
      resourceId: profileId,
      payload: { action: 'publish', profileId, schemas },
    });
    
    // Log activity
    await jobAPI.logActivity({
      type: ActivityType.CONTENT,
      source: ActivitySource.ADMIN,
      userId: getCurrentUser().id,
      action: ActivityAction.PUBLISH,
      description: `Published DCTAP profile: ${profile.shapeLabel}`,
      resourceType: 'dctap_profile',
      resourceId: profileId,
      jobId: job.id,
    });
  }
}
```

```sql
CREATE TABLE dctap_profiles (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  shape_id TEXT NOT NULL,
  shape_label TEXT NOT NULL,
  version TEXT NOT NULL DEFAULT '1.0.0',
  namespace TEXT NOT NULL,
  prefix TEXT,
  description TEXT,
  
  -- Ownership
  created_by TEXT NOT NULL,
  modified_by TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  modified_at TIMESTAMPTZ DEFAULT NOW(),
  
  -- Classification
  site_key TEXT NOT NULL,
  profile_type TEXT NOT NULL,
  parent_profile_id UUID REFERENCES dctap_profiles(id),
  
  -- Status
  status TEXT NOT NULL DEFAULT 'draft',
  published BOOLEAN DEFAULT FALSE,
  deprecated BOOLEAN DEFAULT FALSE,
  
  -- Configuration
  generation_config JSONB DEFAULT '{}',
  
  -- Content
  properties JSONB NOT NULL DEFAULT '[]',
  
  -- Search and indexing
  search_vector tsvector GENERATED ALWAYS AS (
    to_tsvector('english', 
      shape_label || ' ' || 
      COALESCE(description, '') || ' ' ||
      COALESCE(prefix, '')
    )
  ) STORED,
  
  -- Constraints
  UNIQUE(shape_id, site_key, version)
);

-- Indexes
CREATE INDEX idx_dctap_profiles_site_key ON dctap_profiles(site_key);
CREATE INDEX idx_dctap_profiles_type ON dctap_profiles(profile_type);
CREATE INDEX idx_dctap_profiles_status ON dctap_profiles(status);
CREATE INDEX idx_dctap_profiles_search ON dctap_profiles USING GIN(search_vector);
CREATE INDEX idx_dctap_profiles_modified ON dctap_profiles(modified_at DESC);

-- Full-text search function
CREATE OR REPLACE FUNCTION search_dctap_profiles(query_text TEXT)
RETURNS TABLE(
  id UUID,
  shape_label TEXT,
  description TEXT,
  site_key TEXT,
  profile_type TEXT,
  rank REAL
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    p.id,
    p.shape_label,
    p.description,
    p.site_key,
    p.profile_type,
    ts_rank(p.search_vector, plainto_tsquery('english', query_text)) as rank
  FROM dctap_profiles p
  WHERE p.search_vector @@ plainto_tsquery('english', query_text)
    AND p.published = TRUE
  ORDER BY rank DESC;
END;
$$ LANGUAGE plpgsql;

-- RLS policies
ALTER TABLE dctap_profiles ENABLE ROW LEVEL SECURITY;

CREATE POLICY dctap_profiles_read_policy ON dctap_profiles
  FOR SELECT USING (
    published = TRUE OR 
    created_by = auth.uid()::text OR

  );
```
